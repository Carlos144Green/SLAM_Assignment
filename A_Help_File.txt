Start Ros1 Docker:
bash scripts/start.sh ros1

Stereo Kalibr:
rosrun kalibr kalibr_calibrate_cameras \
  --bag ~/data/omni_stereo_20260425_215907Z.bag \
  --topics /stereo/left/color/image_raw /stereo/right/color/image_raw \
  --models omni-radtan omni-radtan \
  --target ~/config/aprilgrid.yaml \
  --bag-from-to 0 50 \
  --approx-sync 0.005
  
  
Intertial Kalibr:
rosrun kalibr kalibr_calibrate_imu_camera   --bag ~/data/omni_stereointertial_20260425_220304Z.bag   --cams ~/data/omni_stereo_20260425_215907Z-camchain-omni-50clipped-approx_005.yaml   --imu ~/config/imu.yaml   --target ~/config/aprilgrid.yaml   --bag-from-to 0 50

How to run bag as video (needs to be in docker env):
1. roscore
2. rosbag play ~/data/omni_stereointertial_20260425_220304Z.bag --clock
3. rosrun rqt_image_view rqt_image_view /stereo/left/color/image_raw

bash scripts/start.sh ros2


ros2 launch stereo_vio stereo_vio.launch.py
